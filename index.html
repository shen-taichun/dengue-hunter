/** * 處理熱點點擊後的邏輯 * 區分：[衛教] 標籤為輔助知識，其餘為必須清除的孳生源 */ function handleSuccessfulHit(spot, x, y) { // 判斷是否為衛教熱點（名稱開頭是否有 [衛教]） const isEducational = spot.name.startsWith("[衛教]"); if (!isEducational) { // --- 處理過關熱點 (孳生源) --- // 避免重複計算 if (!foundHotspots.includes(spot.id)) { foundHotspots.push(spot.id); AudioSynth.playHit(); // 更新進度條與計數 const level = gameData.levels[currentLevelIndex]; const totalCleanable = level.hotspots.filter(h => !h.name.startsWith("[衛教]")).length; const currentFound = foundHotspots.length; document.getElementById('hud-found-count').innerText = currentFound; document.getElementById('hud-progress-bar').style.width = `${(currentFound / totalCleanable) * 100}%`; // 標記已點擊位置 createSuccessMarker(x, y, spot.name); // 檢查是否過關 if (currentFound >= totalCleanable) { setTimeout(() => { handleLevelComplete(); }, 800); } } } else { // --- 處理衛教熱點 (教育用途) --- AudioSynth.playHit(); // 給予點擊音效回饋 // 衛教熱點不執行 foundHotspots.push，因此不會影響進度條 } // 統一顯示提示視窗 (無論哪種熱點都會彈出) const title = isEducational ? "💡 衛教小貼士" : "✨ 發現孳生源！"; const cleanName = spot.name.replace("[衛教]", ""); // 假設您原本有 showAlert 函式，若沒有，請確保視窗顯示邏輯正確 showAlert("success", title, `【${cleanName}】
${spot.desc}`); }
